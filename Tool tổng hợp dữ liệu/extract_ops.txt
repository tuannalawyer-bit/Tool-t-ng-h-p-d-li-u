Sub DownloadOPS_InOut(site)

    ' Changed from MSXML2.XMLHTTP to ServerXMLHTTP which is sometimes more reliable

    Dim http As Object ' We use Object for late binding

    Dim stream As Object

    Dim url As String

    Dim cookie As String

    Dim filePath As String

    

    Dim d1, d2

        

    d2 = Format(Date, "yyyy-mm-dd")

    d1 = Format(Date - 30, "yyyy-mm-dd")



    ' URL is correct

    url = "https://ops.winmart.vn/bao-cao/bao-cao-check-in-out-chi-tiet?app=WinCare&sSite=" & site & "&sites=" & site & "&frda=" & d1 & "&toda=" & d2 & "&emplCode=&p=1&isExport=True"

    Debug.Print url

    

    cookie = shUI.Range("I15").Value

    

    filePath = Environ("USERPROFILE") & "\Downloads\data.xlsx"

    

    ' Use ServerXMLHTTP for potentially better compatibility

    Set http = CreateObject("MSXML2.ServerXMLHTTP")

    

    ' --- Setting up the request ---

    http.Open "GET", url, False

    

    ' 1. Set the Critical Cookie header

    http.setRequestHeader "Cookie", cookie

    

    ' 2. Replicate ALL standard browser headers from your debug request:

    http.setRequestHeader "User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/141.0.0.0 Safari/537.36"

    http.setRequestHeader "Accept", "text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,image/apng,*/*;q=0.8,application/signed-exchange;v=b3;q=0.7"

    http.setRequestHeader "Accept-Language", "en-US,en;q=0.9,vi-VN;q=0.8,vi;q=0.7" ' NEW!

    http.setRequestHeader "Upgrade-Insecure-Requests", "1"

    

    ' The sec-ch-ua headers are included as before

    http.setRequestHeader "sec-ch-ua", """Google Chrome"";v=""141"", ""Not?A_Brand"";v=""8"", ""Chromium"";v=""141"""

    http.setRequestHeader "sec-ch-ua-mobile", "?0"

    http.setRequestHeader "sec-ch-ua-platform", """Windows"""

    

    ' NOTE: Do NOT set "Connection" or "Host" as XMLHTTP handles these.

    

    ' 3. Send the request

    http.send

    

    ' Output for debugging

    Debug.Print "HTTP Status: " & http.Status

    

    ' --- Handling the response ---

    If http.Status = 200 Then

        

        ' Proceed with file saving

        Set stream = CreateObject("ADODB.Stream")

        stream.Type = 1 ' 1 = binary

        stream.Open

        stream.Write http.responseBody

        stream.Position = 0

        stream.SaveToFile filePath, 2 ' 2 = overwrite if exists

        stream.Close

        Debug.Print "? File downloaded successfully to: " & filePath

    Else

        Debug.Print "? Download failed. Status: " & http.Status

    End If

    

    Import_Datafile shInOut, False

    

    ' Clean up objects

    Set http = Nothing

    Set stream = Nothing

End Sub



Function checkDupImei()



    Dim rng As Range

    

    Set rng = shInOut.Range("A1").CurrentRegion

    rng.AdvancedFilter xlFilterCopy, shInOut.Range("M1:M2"), shInOut.Range("M1:N1"), True

    shInOut.Range("M1").CurrentRegion.Sort key1:=shInOut.Range("M1"), order1:=xlAscending, Header:=xlYes

    

    rng.AdvancedFilter xlFilterCopy, shInOut.Range("Q1:Q2"), shInOut.Range("Q1"), True

    rng.AdvancedFilter xlFilterCopy, shInOut.Range("S1:S2"), shInOut.Range("S1"), True

End Function

    If http.Status = 200 Then

        

                

        Dim text

        Dim options

        text = Right(http.responseText, Len(http.responseText) - InStr(http.responseText, "id=""tnid"""))

        options = Split(text, "<option")

        ReDim reArr(1 To UBound(options), 1 To 2)

        r = 1

        For i = LBound(options) To UBound(options)

            If Left(options(i), 6) = " value" Then

                items = Split(options(i), """>")

                tnid = Right(items(0), Len(items(0)) - InStr(items(0), """"))

                If InStr(items(1), "-") > 0 Then

                    site = Left(items(1), InStr(items(1), "-") - 1)

                Else

                    site = ""

                End If

                Debug.Print tnid, site

                reArr(r, 1) = site

                reArr(r, 2) = tnid

                r = r + 1

            End If

        Next i

        

        shTnid.Range("A1").CurrentRegion.Offset(1).ClearContents

        shTnid.Range("A2").Resize(UBound(options), 2).Value = reArr

        

        MsgBox "Done"

    Else

        Debug.Print "? Download failed. Status: " & http.Status

    End If

    

    ' Clean up objects

    Set http = Nothing

    Set stream = Nothing

End Sub
